def train_deploy_model():
    """Train the GPR model on all embedded specimens (used for prediction)."""
    arr = np.array(DATA, dtype=float)
    X, y = arr[:, :7], arr[:, 7]
    kernel = (ConstantKernel(1.0) * Matern(length_scale=[1.0] * 7, nu=1.5)
              + WhiteKernel(1e-3))
    model = Pipeline([("sc", StandardScaler()),
                      ("m", GaussianProcessRegressor(
                          kernel=kernel, alpha=1e-8, normalize_y=True,
                          random_state=42, n_restarts_optimizer=2))])
    model.fit(X, y)
    return model, X


class RoundButton(tk.Canvas):
    """A flat rounded-rectangle button (nicer than the default tk button)."""
    def __init__(self, parent, text, fill, fill_active, command, width=210, height=46):
        super().__init__(parent, width=width, height=height, bg=CARD,
                         highlightthickness=0, bd=0)
        self.command = command
        self.fill, self.fill_active = fill, fill_active
        self.width, self.height = width, height
        self.text = text
        self._draw(fill)
        self.bind("<Button-1>", lambda e: command())
        self.bind("<Enter>", lambda e: self._draw(self.fill_active))
        self.bind("<Leave>", lambda e: self._draw(self.fill))

    def _round_rect(self, x1, y1, x2, y2, r, **kw):
        pts = [x1+r, y1, x2-r, y1, x2, y1, x2, y1+r, x2, y2-r, x2, y2,
               x2-r, y2, x1+r, y2, x1, y2, x1, y2-r, x1, y1+r, x1, y1]
        return self.create_polygon(pts, smooth=True, **kw)

    def _draw(self, fill):
        self.delete("all")
        self._round_rect(1, 1, self.width-1, self.height-1, 12, fill=fill, outline=fill)
        self.create_text(self.width/2, self.height/2, text=self.text,
                         fill="white", font=("Segoe UI", 12, "bold"))

    # ---------- gauge drawing ----------
    def _draw_gauge(self, fc):
        g = self.gauge
        g.delete("all")
        x0, x1, yb = 4, 296, 30
        ymin, ymax = 24.0, 40.0  # display range of the target
        g.create_line(x0, yb, x1, yb, fill="#d9d9d9", width=6, capstyle="round")
        frac = max(0.0, min(1.0, (fc - ymin) / (ymax - ymin)))
        px = x0 + frac * (x1 - x0)
        g.create_line(x0, yb, px, yb, fill="#000000", width=6, capstyle="round")
        g.create_oval(px-7, yb-7, px+7, yb+7, fill="#000000", outline="white", width=2)
        g.create_text(x0, yb+16, text="24", fill=INK, font=("Segoe UI", 10, "bold"), anchor="w")
        g.create_text(x1, yb+16, text="40 MPa", fill=INK, font=("Segoe UI", 10, "bold"), anchor="e")

    # ---------- actions ----------
    def calculate(self):
        vals = []
        for i, (_, name, _, lo, hi, _) in enumerate(FEATURES):
            raw = self.entries[name].get().strip()
            try:
                v = float(raw)
            except ValueError:
                messagebox.showerror("Invalid input",
                                     f"Please enter a valid number for {name}.")
                return
            vals.append(v)
        fc = float(self.model.predict(np.array([vals]))[0])
        self.pred_var.set(f"{fc:.2f}")
        self._draw_gauge(fc)

    def reset(self):
        for (_, name, _, _, _, _) in FEATURES:
            self.entries[name].delete(0, "end")
        self.pred_var.set("--")
        self.gauge.delete("all")

    def load_example(self):
        for (_, name, _, _, _, default) in FEATURES:
            self.entries[name].delete(0, "end")
            self.entries[name].insert(0, f"{default:g}")
        self.calculate()
